function getAttendese() { var selectedRollNumbers = []; var unselectedRollNumbers = []; $(".student").each(function (index) { var rollNumber = $(this).attr("class").split(" ")[0]; var displayNumber; // Lateral Entry students if (rollNumber.substring(0, 2) == "24") { var leIndex = 1; $(".student").each(function (i) { if (i >= index) { return false; } var previousRoll = $(this).attr("class").split(" ")[0]; if (previousRoll.substring(0, 2) == "24") { leIndex++; } }); displayNumber = "LE-" + leIndex.toString().padStart(2, "0"); } // Regular students else { var regularIndex = 1; $(".student").each(function (i) { if (i >= index) { return false; } var previousRoll = $(this).attr("class").split(" ")[0]; if (previousRoll.substring(0, 2) != "24") { regularIndex++; } }); displayNumber = regularIndex.toString().padStart(2, "0"); } if ($(this).hasClass("P")) { selectedRollNumbers.push(displayNumber); } else if ($(this).hasClass("A")) { unselectedRollNumbers.push(displayNumber); } }); return { selectedRollNumbers, unselectedRollNumbers }; }
